# Getting Started

This guide shows the minimal setup for the **NestJS CRUD Config 5** contract: Automator 5, NestJS 12, TypeORM 1, ES modules only.

## Installation

Use the 5.x range below. It also accepts the 5.0.0 prereleases.

## Peer dependencies

CrudConfig 5 needs Node.js `^22.13.0 || >=24.11.0`. Install the required peer dependencies (if not already installed in your app):

```bash copy
npm install @elsikora/nestjs-crud-automator@^5.0.0-0 @nestjs/common@^12 @nestjs/core@^12 @nestjs/typeorm@^12 typeorm@^1
npm install @nestjs/passport@^12 @nestjs/platform-fastify@^12 @nestjs/swagger@^12 @nestjs/throttler class-transformer class-validator fastify lodash
```

The package ships as ES modules only. ES module applications import it with `moduleResolution` `bundler`, `node16` or `nodenext`; CommonJS applications keep `require("@elsikora/nestjs-crud-config")`, which Node.js loads through `require(esm)`. Importing the package loads no `.env` file: `NCC_LOG_LEVEL` comes from the application's environment.

## Automator 5 compatibility

CrudConfig 5 requires `@elsikora/nestjs-crud-automator ^5.0.0-dev.5`; Automator 4 and older are not supported. Route overrides continue to use Automator generation config:

```typescript copy
DELETE: {
 generation: {
  isEnabled: false,
 },
},
```

CrudConfig's generated services use inherited `@ApiService` CRUD methods for the same dynamic entity, which satisfies the Automator generated-route capability boundary without a consumer wrapper. Generated routes run Automator's `ApiAuthorizationGuard`, so an application with the section or data controllers enabled imports Automator's global `ApiAuthorizationModule` once in its root module, as the examples below do. Transaction ownership, nested ConfigData relation input, and the EventEmitter-free runtime are unchanged. Existing applications should follow [Migrating to 5.0](/docs/nestjs-crud-config/guides/migrating-to-5-0).

## Basic setup (sync registration)

Register the module first (to create dynamic entities), then register TypeORM using the exported entity tokens.

```typescript filename="app.module.ts" copy
import { Module } from "@nestjs/common";
import { TypeOrmModule } from "@nestjs/typeorm";
import { ApiAuthorizationModule } from "@elsikora/nestjs-crud-automator";
import { CrudConfigModule, TOKEN_CONSTANT } from "@elsikora/nestjs-crud-config";

@Module({
 imports: [
  // Automator's generated routes run ApiAuthorizationGuard; import this global module once in the root module
  ApiAuthorizationModule,
  CrudConfigModule.register({
   environment: "development",
   shouldAutoCreateSections: true,

   cacheOptions: {
    isEnabled: true,
    maxCacheItems: 1000,
    maxCacheTTL: 3600000,
   },

   encryptionOptions: {
    isEnabled: true,
    encryptionKey: process.env.CONFIG_ENCRYPTION_KEY,
   },

   controllersOptions: {
    section: { isEnabled: true, properties: { path: "api/config/sections" } },
    data: { isEnabled: true, properties: { path: "api/config/data" } },
   },

   entityOptions: {
    tablePrefix: "app_",
    configSection: { tableName: "config_sections" },
    configData: { tableName: "config_data" },
   },
  }),

  TypeOrmModule.forRootAsync({
   imports: [CrudConfigModule],
   inject: [
    TOKEN_CONSTANT.CONFIG_SECTION_ENTITY,
    TOKEN_CONSTANT.CONFIG_DATA_ENTITY,
    TOKEN_CONSTANT.CONFIG_MIGRATION_ENTITY,
   ],
   useFactory: async (sectionEntity, dataEntity, migrationEntity) => ({
    type: "postgres",
    // ... connection options ...
    entities: [sectionEntity, dataEntity, migrationEntity],
    synchronize: true,
   }),
  }),
 ],
})
export class AppModule {}
```

## Basic setup (async registration)

When using `registerAsync()`, static options (controllers and entity customization) must be provided via `staticOptions`.

```typescript filename="app.module.ts" copy
import { Module } from "@nestjs/common";
import { ConfigModule, ConfigService } from "@nestjs/config";
import { ApiAuthorizationModule } from "@elsikora/nestjs-crud-automator";
import { CrudConfigModule } from "@elsikora/nestjs-crud-config";

@Module({
 imports: [
  ApiAuthorizationModule,
  ConfigModule.forRoot(),
  CrudConfigModule.registerAsync({
   imports: [ConfigModule],
   inject: [ConfigService],
   useFactory: async (configService: ConfigService) => ({
    environment: configService.get("NODE_ENV", "development"),
    encryptionOptions: {
     isEnabled: true,
     encryptionKey: configService.get("ENCRYPTION_KEY"),
    },
    cacheOptions: {
     isEnabled: true,
     maxCacheItems: 1000,
     maxCacheTTL: 3600000,
    },
   }),
   staticOptions: {
    controllersOptions: {
     section: { isEnabled: true, properties: { path: "api/config/sections" } },
     data: { isEnabled: true, properties: { path: "api/config/data" } },
    },
    entityOptions: {
     tablePrefix: "app_",
     configSection: { tableName: "config_sections" },
     configData: { tableName: "config_data" },
    },
    migrationEntityOptions: {
     tableName: "config_migrations",
    },
   },
  }),
 ],
})
export class AppModule {}
```
